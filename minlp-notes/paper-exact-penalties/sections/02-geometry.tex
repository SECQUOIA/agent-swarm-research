\section{Exactness notions and residual geometry}\label{sec:geometry}
This section fixes the setting and the vocabulary of the paper: three
notions of exactness and two thresholds. It then proves three tools.
Balanced mixtures describe the optimized dual value in primal terms and
supply the lower-bound witnesses of Sections~\ref{sec:lower},
\ref{sec:calibration} and~\ref{sec:perturbation}; one relaxed equality
admits explicit thresholds; and a slice-wise sufficient condition at
multiplier zero underlies the upper bounds of Sections~\ref{sec:upper}
and~\ref{sec:fixedcount}.

\subsection{Setting}
Let $X$ be a nonempty compact subset of a finite-dimensional Euclidean
space, and let $f:X\to\R$ and $r:X\to\R^m$ be continuous. The set $X$
is the \emph{native set}: it collects the \emph{native constraints},
which are kept in the penalized subproblem. The equalities $r(x)=0$ are
the \emph{relaxed equalities}; they are moved into the penalty. In
applications they are linking or coupling constraints, so we also call
them linking equalities. In Sections~\ref{sec:upper}
and~\ref{sec:fixedcount} they have the form $r(x,z)=Ax+Bz-b$. We assume that the feasible set
$F=\{x\in X:r(x)=0\}$ is nonempty. Fix a norm $\norm{\cdot}$ on $\R^m$,
with dual norm $\norm{\cdot}_*$. For $\rho\ge0$ and $\lambda\in\R^m$ let
\begin{equation}\label{eq:model}
 v=\min_{x\in F}f(x),\qquad
 L_\rho(\lambda)=\min_{x\in X}\{f(x)+\lambda^Tr(x)+\rho\norm{r(x)}\},
 \qquad D_\rho=\sup_{\lambda\in\R^m}L_\rho(\lambda).
\end{equation}
Here $v$ is the optimal value, $L_\rho(\lambda)$ is the penalized value
at the multiplier $\lambda$, and $D_\rho$ is the optimized (dual) value.
The minima exist because $F$ is closed in the compact set $X$ and $f$
and $r$ are continuous. As a minimum of functions that are affine in
$\lambda$, $L_\rho$ is concave and finite on $\R^m$, hence continuous.
On $F$ the penalized objective equals $f$, so weak duality holds:
\[
 L_\rho(\lambda)\le D_\rho\le v\qquad\text{for all }\rho\ge0,\ \lambda\in\R^m.
\]
Thus $D_\rho$ is the supremum of the lower bounds on $v$ that the
multipliers give at the coefficient $\rho$.

\paragraph{Slices.}
In mixed-integer models some coordinates of the points of $X$ are
integer variables; Sections~\ref{sec:upper} and~\ref{sec:fixedcount}
write a point as $(x,z)$ with continuous part $x$ and integer part $z$.
The \emph{slice} $X_z$ is the set of native points whose integer part
equals $z$. When convenient we identify $X_z$ with the set of continuous
parts of its points. Because $X$ is compact, only finitely many slices
are nonempty, and each of them is compact. A slice is
\emph{equality-feasible} if it meets $F$, that is, if it contains a
point with $r=0$; otherwise it is \emph{equality-infeasible}. A nonempty
equality-infeasible slice has positive \emph{residual separation}
$\delta_z=\min_{x\in X_z}\norm{r(x)}$, by compactness and continuity.

\paragraph{Encoding conventions.}
Statements about encoding length use ordinary binary encoding. A
rational number is encoded by the binary representations of its integer
numerator and its positive integer denominator. An instance is given by
explicit rational coefficients, and $N$ denotes its binary encoding
length. A sparse encoding records the variable indices as well as the
nonzero coefficients; Assumption~\ref{ass:model} fixes the encoding of
quadratic data in Sections~\ref{sec:upper} and~\ref{sec:fixedcount}.
The norm and the relaxed equalities are fixed parts of the formulation;
rescaling $r$ or changing the norm gives a different instance, which can
have different thresholds (Section~\ref{sec:lower-scope}).

\subsection{Exactness notions and thresholds}
\begin{definition}[Exactness notions and thresholds]\label{def:exactness}
Let $\rho\ge0$ and $\lambda\in\R^m$.
\begin{itemize}
\item $\rho$ is \emph{value-exact} if $D_\rho=v$.
\item $\rho$ is \emph{exact at $\lambda$} if $L_\rho(\lambda)=v$.
\item $\rho$ is \emph{minimizer-exact at $\lambda$} if the set of
 minimizers of $f+\lambda^Tr+\rho\norm{r}$ over $X$ equals
 $\argmin_{x\in F}f(x)$.
\end{itemize}
The \emph{least penalty} is $\rho_*=\inf\{\rho\ge0:D_\rho=v\}$. The
\emph{threshold at $\lambda$} is
$\rho_*(\lambda)=\inf\{\rho\ge0:L_\rho(\lambda)=v\}$, and $\rho_*(0)$ is
the \emph{zero-multiplier threshold}. We use the convention
$\inf\varnothing=+\infty$.
\end{definition}
Minimizer-exactness is what primal recovery needs: every minimizer of the penalized subproblem is then an
optimal solution of the original problem.

\paragraph{The threshold at a fixed multiplier.}
For every $\lambda\in\R^m$,
\begin{equation}\label{eq:fixed}
 \rho_*(\lambda)=\max\left\{0,\sup_{x\in X\setminus F}
 \frac{v-f(x)-\lambda^Tr(x)}{\norm{r(x)}}\right\},
\end{equation}
where the supremum of the empty set is $-\infty$. Indeed, by weak
duality, $L_\rho(\lambda)=v$ holds exactly when
$f(x)+\lambda^Tr(x)+\rho\norm{r(x)}\ge v$ for every $x\in X$. On $F$
this inequality holds because $f\ge v$ there. For $x\notin F$ we have
$\norm{r(x)}>0$, and the inequality says that $\rho$ is at least the
ratio in \eqref{eq:fixed}. Two consequences follow.
\begin{itemize}
\item The coefficients exact at $\lambda$ form the interval
 $[\rho_*(\lambda),\infty)$ when $\rho_*(\lambda)$ is finite. The
 threshold $\rho_*(\lambda)$ may be $+\infty$; then no coefficient is
 exact at $\lambda$ (Example~\ref{ex:onesided}).
\item Every $\rho>\rho_*(\lambda)$ is minimizer-exact at $\lambda$: at
 each $x\notin F$, the extra term $(\rho-\rho_*(\lambda))\norm{r(x)}$
 is positive, so the penalized objective exceeds $v$, while on $F$ it
 equals $f$.
\end{itemize}

\paragraph{Implications between the notions.}
For every $\rho\ge0$ and $\lambda\in\R^m$,
\[
 \text{minimizer-exact at }\lambda\ \Longrightarrow\
 \text{exact at }\lambda\ \Longrightarrow\ \text{value-exact}.
\]
For the first implication, a point of $\argmin_{x\in F}f(x)$ is then a
penalized minimizer, with penalized value $v$. For the second,
$v=L_\rho(\lambda)\le D_\rho\le v$; in this case $\lambda$ attains the
dual supremum. Consequently $\rho_*\le\rho_*(\lambda)$ for every
$\lambda$. Neither converse holds. Example~\ref{ex:onesided} and
Appendix~\ref{app:attainment} give value-exact coefficients that are
exact at no multiplier, because the dual supremum is not attained.
Exactness at $\lambda$ does not imply minimizer-exactness at $\lambda$
either: at $\rho=\rho_*(\lambda)$, infeasible points can attain the
value $v$. Such ties occur in Sections~\ref{sec:lower}
and~\ref{sec:calibration}.

\paragraph{Relation to the sharp Lagrangian.}
The function $L_\rho(\lambda)$ is the dual function of the sharp
Lagrangian \citep[Example~11.58]{RockafellarWets1998}. For compact $X$
their level-boundedness condition holds. Then $\lambda$ supports an
exact penalty representation in the sense of
\citet[Definition~11.60]{RockafellarWets1998} exactly when
$\rho_*(\lambda)$ is finite, and the adequate penalty thresholds are
then the positive $\rho\ge\rho_*(\lambda)$
\citep[Theorem~11.61]{RockafellarWets1998}. For $\lambda=0$ this is
calmness from below of the value function, that is, of the optimal value
as a function of the right-hand side of the relaxed equalities
\citep[Example~11.62(b)]{RockafellarWets1998}. The least penalty
$\rho_*$ takes the multiplier into account, and it can be smaller than
every $\rho_*(\lambda)$ (Example~\ref{ex:onesided}).

\subsection{Balanced mixtures and the least penalty}
A \emph{balanced mixture} $\pi$ is a finite family of native points
$x_i\in X$ with weights $\theta_i\ge0$ such that
\[
 \sum_i\theta_i=1\qquad\text{and}\qquad\sum_i\theta_ir(x_i)=0.
\]
Its \emph{average cost} and \emph{average residual norm} are
\[
 c_\pi=\sum_i\theta_if(x_i),\qquad
 s_\pi=\sum_i\theta_i\norm{r(x_i)}.
\]
Let $\mathcal B$ be the set of all balanced mixtures. The residuals
cancel on average, so the multiplier term $\lambda^Tr$ averages to zero
over a balanced mixture. Hence, for every $\lambda$,
\[
 L_\rho(\lambda)\le\sum_i\theta_i
 \bigl[f(x_i)+\lambda^Tr(x_i)+\rho\norm{r(x_i)}\bigr]=c_\pi+\rho s_\pi .
\]
A balanced mixture with $c_\pi+\rho s_\pi<v$ therefore certifies a
duality gap that no multiplier can remove. Proposition~\ref{prop:mixture}
shows that the best such certificate gives exactly $D_\rho$. The
lower-bound witnesses in Sections~\ref{sec:lower}, \ref{sec:calibration}
and~\ref{sec:perturbation} are balanced mixtures.

The convexification argument behind Proposition~\ref{prop:mixture} is
that of \citet{BolandEberhard2015} and \citet{FeizollahiAhmedSun2017};
\citet[Theorem~1]{FeizollahiAhmedSun2017} use a lifted convex hull under
a relative-interior condition. The residual-space version below, for
compact $X$, needs no such condition and does not assume that the dual
supremum is attained.

\begin{proposition}[Balanced mixtures]\label{prop:mixture}
For every $\rho\ge0$, the optimized value $D_\rho$ is the minimum of
$c_\pi+\rho s_\pi$ over all balanced mixtures $\pi$, and this minimum is
attained by a balanced mixture of at most $m+1$ points:
\begin{equation}\label{eq:mixture}
 D_\rho=\min\left\{\sum_{i=1}^{m+1}\theta_i
 \bigl[f(x_i)+\rho\norm{r(x_i)}\bigr]:
 \begin{array}{l}
 x_i\in X,\ \theta_i\ge0,\\
 \sum_i\theta_i=1,\quad\sum_i\theta_i r(x_i)=0
 \end{array}\right\}.
\end{equation}
\end{proposition}
\begin{proof}
We show that $D_\rho$ is the lowest point above the origin of a convex
hull in $\R^m\times\R$, and then reduce the number of points.

Let $\mathcal G=\{(r(x),f(x)+\rho\norm{r(x)}):x\in X\}\subseteq\R^m\times\R$.
It is compact as a continuous image of $X$, so its convex hull
$\mathcal C=\conv\mathcal G$ is compact. The points of $\mathcal C$ of
the form $(0,t)$ are exactly the points $(0,c_\pi+\rho s_\pi)$ with
$\pi\in\mathcal B$. Such points exist, since a point of $F$ is a
balanced mixture by itself. Let $w=\min\{t:(0,t)\in\mathcal C\}$. Then
$c_\pi+\rho s_\pi\ge w$ for every $\pi\in\mathcal B$, with equality for
some $\pi$.

First, $D_\rho\le w$: averaging over a balanced mixture with value $w$
cancels $\lambda$, so $L_\rho(\lambda)\le w$ for every $\lambda$.

Second, $D_\rho\ge w$. Let $a<w$. Then $(0,a)\notin\mathcal C$, and
strict separation of the point $(0,a)$ from the compact convex set
$\mathcal C$ gives $(u,\beta)\in\R^m\times\R$ with
\[
 u^T\zeta+\beta t>\beta a\qquad\text{for all }(\zeta,t)\in\mathcal C.
\]
At $(\zeta,t)=(0,w)$ this reads $\beta w>\beta a$, so $\beta>0$.
At the points of $\mathcal G$, after division by $\beta$, it reads
$f(x)+(u/\beta)^Tr(x)+\rho\norm{r(x)}>a$ for every $x\in X$. The
minimum over the compact set $X$ is attained, so $L_\rho(u/\beta)>a$
and $D_\rho>a$. Letting $a\uparrow w$ gives $D_\rho=w$. No attaining
multiplier was used.

Finally, the support. By Carath\'eodory's theorem in $\R^{m+1}$, the
point $(0,w)$ is a convex combination of at most $m+2$ points of
$\mathcal G$; take one with positive weights $\theta_i$ at points
$x_i$. If there are $m+2$ points, the vectors $(1,r(x_i))\in\R^{m+1}$
are linearly dependent: some $\xi\ne0$ satisfies $\sum_i\xi_i=0$
and $\sum_i\xi_ir(x_i)=0$. Because its entries sum to zero, $\xi$
has a negative entry. For every small $|\nu|$, the weights
$\theta+\nu\xi$ again form a balanced mixture, whose value
$\sum_i(\theta_i+\nu\xi_i)[f(x_i)+\rho\norm{r(x_i)}]$ is affine
in $\nu$, at least $w$, and equal to $w$ at $\nu=0$. Hence the
value does not depend on $\nu$. Increase $\nu$ from zero until
the first weight with $\xi_i<0$ vanishes. The value stays $w$ and at
most $m+1$ points remain. Padding with zero weights gives the form
\eqref{eq:mixture}.
\end{proof}
The average in \eqref{eq:mixture} contains the norms of the individual
residuals, not the norm of their average, which is zero.

The support bound $m+1$ is sharp. Let $X$ consist of $m+2$ points: one
with $(r,f)=(0,0)$, and $m+1$ points with cost $-1$ and residuals
$e_1,\ldots,e_m$ and $-\sum_je_j$, where $e_j$ are the unit vectors.
At $\rho=0$ a balanced mixture must give equal weight to the last $m+1$
points, so its value is $-1$ times their total weight. The optimal
mixture gives weight $1/(m+1)$ to each of them, with value $-1$, while
every balanced mixture of at most $m$ points has value $0$.

\begin{corollary}\label{cor:ratio}
The least penalty is
\begin{equation}\label{eq:ratio}
 \rho_*=\max\left\{0,\sup_{\pi\in\mathcal B:\,s_\pi>0}
                  \frac{v-c_\pi}{s_\pi}\right\},
\end{equation}
where the supremum of the empty set is $-\infty$. The supremum may be
restricted to balanced mixtures of at most $m+1$ points. If $\rho_*$ is
finite, it is value-exact.
\end{corollary}
\begin{proof}
By Proposition~\ref{prop:mixture}, $D_\rho\ge v$ if and only if
$c_\pi+\rho s_\pi\ge v$ for all $\pi\in\mathcal B$, and also if and only
if this holds for all balanced mixtures of at most $m+1$ points. A
mixture with $s_\pi=0$ gives positive weight only to points of $F$, so
$c_\pi\ge v$ and its condition always holds. For $s_\pi>0$ the condition
reads $\rho\ge(v-c_\pi)/s_\pi$. Since $D_\rho\le v$ always, $D_\rho=v$
holds exactly when $\rho\ge0$ is at least every such ratio. If the
supremum of the ratios is less than $+\infty$, the least such $\rho$ is
\eqref{eq:ratio}, and it satisfies all the conditions; if the supremum
is $+\infty$, no $\rho$ does.
\end{proof}
Each condition $c_\pi+\rho s_\pi\ge v$ is closed and monotone in $\rho$.
Hence, when $\rho_*$ is finite, the value-exact coefficients form the
interval $[\rho_*,\infty)$, just as the coefficients exact at $\lambda$
form $[\rho_*(\lambda),\infty)$. In words, $\rho_*$ is the supremum of the objective gain $v-c_\pi$ per
unit of average residual norm over balanced mixtures with $s_\pi>0$, or
zero if this supremum is negative or no such mixture exists. In \eqref{eq:fixed} the ratio is taken over
single points instead, and it includes the multiplier term.

\subsection{One relaxed equality}
For one relaxed equality the least penalty and the thresholds have
closed forms in terms of two one-sided rates.

\begin{proposition}[One relaxed equality]\label{prop:scalar}
Let $m=1$ with the absolute value as norm, and suppose that $X$ contains
points with $r(x)>0$ and points with $r(x)<0$. Define
\[
 A_+=\sup_{x\in X:\,r(x)>0}\frac{v-f(x)}{r(x)},\qquad
 A_-=\sup_{x\in X:\,r(x)<0}\frac{v-f(x)}{-r(x)}.
\]
Then $\rho_*(\lambda)=\max\{0,A_+-\lambda,A_-+\lambda\}$ for every
$\lambda\in\R$; in particular, the zero-multiplier threshold is
$\rho_*(0)=\max\{0,A_+,A_-\}$. If $A_+$ and $A_-$ are finite, then for
all $\rho\ge0$ and $\lambda\in\R$,
\begin{equation}\label{eq:scalar}
 \rho_*=\max\{0,(A_++A_-)/2\},\qquad
 L_\rho(\lambda)=v\ \Longleftrightarrow\
 A_+-\rho\le\lambda\le\rho-A_-.
\end{equation}
If $A_+$ or $A_-$ is $+\infty$, no finite $\rho$ is value-exact.
\end{proposition}
\begin{proof}
The fixed-multiplier part is \eqref{eq:fixed} split by the sign of the
residual; the optimized part follows once we show that an exact dual
value is attained.

For $r(x)>0$ the ratio in \eqref{eq:fixed} is $(v-f(x))/r(x)-\lambda$,
and for $r(x)<0$ it is $(v-f(x))/(-r(x))+\lambda$. Their suprema are
$A_+-\lambda$ and $A_-+\lambda$, which gives the formula for
$\rho_*(\lambda)$. Equivalently, $L_\rho(\lambda)=v$ if and only if
\[
 (\rho+\lambda)\,r(x)\ge v-f(x)\ \text{ whenever }r(x)>0,\qquad
 (\rho-\lambda)\,|r(x)|\ge v-f(x)\ \text{ whenever }r(x)<0.
\]
When $A_+$ and $A_-$ are finite, these conditions are the interval in
\eqref{eq:scalar}.

Next, if $D_\rho=v$, some multiplier attains it. Fix points $x_+$ and
$x_-$ with $r(x_+)>0$ and $r(x_-)<0$. Then
$L_\rho(\lambda)\le f(x_+)+(\lambda+\rho)r(x_+)$, which tends to
$-\infty$ as $\lambda\to-\infty$, and
$L_\rho(\lambda)\le f(x_-)+(\rho-\lambda)|r(x_-)|$, which tends to
$-\infty$ as $\lambda\to+\infty$. Hence every sequence $\lambda_j$ with
$L_\rho(\lambda_j)\to v$ is bounded. A subsequence converges to some
$\bar\lambda$, and continuity of $L_\rho$ gives $L_\rho(\bar\lambda)=v$.

Therefore $D_\rho=v$ if and only if
$\rho\ge\max\{0,A_+-\lambda,A_-+\lambda\}$ for some $\lambda$. If $A_+$
or $A_-$ is $+\infty$, this never holds. Otherwise the interval in
\eqref{eq:scalar} is nonempty exactly when $A_+-\rho\le\rho-A_-$, that
is, when $\rho\ge(A_++A_-)/2$. Together with $\rho\ge0$ this gives
$\rho_*$.
\end{proof}
In words: a point with positive residual is charged $\rho+\lambda$ per
unit of residual, and a point with negative residual is charged
$\rho-\lambda$. A multiplier moves charge from one side to the other but
cannot change the total $2\rho$. The two sides need at least $A_+$ and
$A_-$ per unit, respectively. So the optimized threshold is the average
$(A_++A_-)/2$ of the two one-sided rates, while multiplier zero needs
their maximum. A rate can be negative: $A_+<0$ means that, for some $c>0$, every point
with $r(x)>0$ costs at least $v+c\,r(x)$, and similarly for $A_-$. When both are nonnegative,
$\rho_*(0)=\max\{A_+,A_-\}\le A_++A_-=2\rho_*$: optimizing the
multiplier reduces the required coefficient by at most a factor two.
When $(A_++A_-)/2\ge0$, the interval in \eqref{eq:scalar} at
$\rho=\rho_*$ consists of the single multiplier $\lambda=(A_+-A_-)/2$.

For $m=1$, if residuals occur on only one side of zero, every balanced
mixture is supported on $F$: its weighted residuals have one sign and sum
to zero. Hence $c_\pi\ge v$ for every $\pi\in\mathcal B$, so
Proposition~\ref{prop:mixture} gives $D_\rho=v$ for all $\rho\ge0$, and
$\rho_*=0$. Attainment can nevertheless fail.

\begin{example}[Value-exact, but exact at no multiplier]\label{ex:onesided}
Let $X=[0,1]$, $r(x)=x$ and $f(x)=-\sqrt x$, with the absolute value as
norm. Then $F=\{0\}$ and $v=0$. All residuals are nonnegative, so
$D_\rho=0=v$ for every $\rho\ge0$. Fix $\lambda$ and $\rho$, and put
$t=\lambda+\rho$. For every $x\in(0,1]$ with $tx<\sqrt x$, which holds
for all small $x>0$, the penalized objective $-\sqrt x+tx$ is negative.
Hence $L_\rho(\lambda)<0=v$ for every finite $\lambda$ and $\rho$: the
threshold $\rho_*(\lambda)$ is $+\infty$ for every $\lambda$, although
$\rho_*=0$. Appendix~\ref{app:attainment} records the same phenomenon in
a mixed-integer example with a convex quadratic constraint from the
literature.
\end{example}

\subsection{A sufficient condition from integer slices}
The next lemma is the template for the upper bounds of
Sections~\ref{sec:upper} and~\ref{sec:fixedcount}. It reduces exactness
at multiplier zero to two separate quantities: a multiplier bound on
each equality-feasible slice and the residual separation of each
equality-infeasible slice.

\begin{lemma}[A bound from integer slices]\label{lem:slices}
Suppose that $X$ is the union of finitely many nonempty compact sets
$X_z$ (for example, the nonempty slices of a mixed-integer native set).
Let $\underline f$ and $\overline f$ be numbers with
$f\ge\underline f$ on $X$ and $v\le\overline f$, and let $\Lambda\ge0$.
Suppose that for every $X_z$ that meets $F$ there is
$\lambda_z\in\R^m$ with $\norm{\lambda_z}_*\le\Lambda$ and
\[
 f(x)+\lambda_z^Tr(x)\ge v_z:=\min_{x'\in X_z\cap F}f(x')
 \qquad\text{for all }x\in X_z ,
\]
and that some $\delta>0$ satisfies $\min_{x\in X_z}\norm{r(x)}\ge\delta$
for every $X_z$ that does not meet $F$. Then every $\rho$ with
\begin{equation}\label{eq:slicebound}
 \rho\ge\max\{\Lambda,(\overline f-\underline f)/\delta\}
\end{equation}
is exact at multiplier zero, and every $\rho$ strictly larger than this
maximum is minimizer-exact at multiplier zero. If every $X_z$ meets
$F$, the second term in \eqref{eq:slicebound} may be omitted.
\end{lemma}
\begin{proof}
Let $\rho$ satisfy \eqref{eq:slicebound} and let $x\in X_z$. If $X_z$
meets $F$, then $\lambda_z^Tr(x)\le\norm{\lambda_z}_*\norm{r(x)}\le
\rho\norm{r(x)}$, so
\[
 f(x)+\rho\norm{r(x)}\ge f(x)+\lambda_z^Tr(x)\ge v_z\ge v .
\]
If $X_z$ does not meet $F$, then
$f(x)+\rho\norm{r(x)}\ge\underline f+\rho\delta\ge\overline f\ge v$.
Thus $L_\rho(0)\ge v$, and weak duality gives $L_\rho(0)=v$. Hence
$\rho_*(0)$ is at most the maximum in \eqref{eq:slicebound}, and every
strictly larger coefficient is minimizer-exact at multiplier zero by the
consequences of \eqref{eq:fixed}.
\end{proof}
The two terms in \eqref{eq:slicebound} control different parts of the
penalty. The multiplier bound $\Lambda$ concerns equality-feasible
slices: since $f(x)\ge v_z-\Lambda\norm{r(x)}$ on such a slice, it
bounds how fast the objective can improve when the slice leaves the
relaxed equalities. The separation $\delta$ concerns
equality-infeasible slices: it bounds how close a slice that cannot
satisfy the relaxed equalities comes to satisfying them, which matters
when such a slice contains improving points. When $f$ and the native
constraints are convex on each slice and $r$ is affine, Slater-type
regularity on equality-feasible slices, such as the refined
Slater condition of Assumption~\ref{ass:model}, supplies the
multipliers $\lambda_z$ and hence a finite $\Lambda$; it says nothing
about $\delta$. The chain of Section~\ref{sec:lower} shows that
$\delta$ can be the bottleneck: its slices have strict points with slack
$1/8$ and $\lambda_z=0$ works on its equality-feasible slice, yet its
equality-infeasible slice has separation $\delta_d=2^{-2^d}$, and the
lemma's coefficient $1/\delta_d$ is exactly its zero-multiplier
threshold.
