\section{Objective distributions and the amount of primal movement}
\label{sec:distribution}
This section takes all scales equal to one. Pool the $r$ positive
objective weights and order them as $w_1\geq\cdots\geq w_r>0$. The
objective gap $E(\eta)$ of the central point and the squared central
speed in the logarithmic parameter are
\begin{equation}
 E(\eta)=\sum_iw_i(1-q(\eta w_i)),\qquad
 r_{\mathrm{eff}}(\eta)=\sum_i\left(1-\frac1{\sqrt{1+(\eta w_i)^2}}\right).
 \label{eq:effective-rank}
\end{equation}
Thus the primal speed depends only on the active channels.
For a matrix factor, $r_{\mathrm{eff}}=\tr[I-(I+\eta^2CC^*)^{-1/2}]$;
for products the traces add. Inactive channels contribute zero even
if the barrier parameter of the domain is large.

\begin{proposition}[Gap and path length in terms of the weights]
\label{prop:distribution-law}
Let $Q_j(\eta)=\sum_i\min((\eta w_i)^j,1)$ for $j=1,2$, and set
$v_0=\sqrt{1-1/\sqrt2}$. Then
\begin{align}
 (2-\sqrt2)Q_1(\eta)/\eta&\leq E(\eta)\leq Q_1(\eta)/\eta,
 \label{eq:Q1}\\
 v_0\int_{\eta_0}^{\eta_1}\sqrt{Q_2(\eta)}\frac{\dd\eta}{\eta}
 &\leq L_\CP(\eta_0,\eta_1)
 \leq\int_{\eta_0}^{\eta_1}\sqrt{Q_2(\eta)}\frac{\dd\eta}{\eta}.
 \label{eq:Q2}
\end{align}
Here $\eta>0$ in \eqref{eq:Q1} and $0\leq\eta_0<\eta_1$ in
\eqref{eq:Q2}; for $\eta_0=0$ the improper integral converges.
\end{proposition}
\begin{proof}
The identity $z(1-q(z))=1+z-\sqrt{1+z^2}$ gives
\[
 (2-\sqrt2)\min(z,1)\leq z(1-q(z))\leq\min(z,1).
\]
For $0<z\leq1$, divide by $z$ and use that $q$ increases; for $z\geq1$
use monotonicity of $1+z-\sqrt{1+z^2}$ and its limit one.
Furthermore,
\[
 1-(1+z^2)^{-1/2}
 =\frac{z^2}{\sqrt{1+z^2}(1+\sqrt{1+z^2})}
 \in[(1-1/\sqrt2)\min(z^2,1),\,\min(z^2,1)].
\]
For $z\leq1$ bound its displayed denominator between $2$ and
$2+\sqrt2$; for $z\geq1$ use monotonicity. Sum and integrate.
\end{proof}
The integral in \eqref{eq:Q2} has a closed form between consecutive
breakpoints $1/w_i$. If $m$ channels have $\eta w_i\geq1$ and
$B=\sum_{i>m}w_i^2$, the integrand is $\sqrt{m+B\eta^2}/\eta$.
For $m,B>0$ an antiderivative is
\[
 \sqrt{m+B\eta^2}+\sqrt m\log
 \frac{\sqrt B\eta}{\sqrt m+\sqrt{m+B\eta^2}}.
\]
For $m=0$ it is $\sqrt B\eta$; for $B=0$ it is $\sqrt m\log\eta$.
Thus the bounds need only the weights, not a path in the matrix
space.

\begin{corollary}[Central length bounds from the $m$ largest weights]
\label{cor:numerical-rank}
Assume $w_1\leq1$ and $0<\eps\leq1$. If $1\leq m\leq r$ and
$\sum_{i>m}w_i\leq\eps/2$, then the central point at $\eta_f=2m/\eps$ is
$\eps$-accurate and
\begin{equation}
 L_\CP(\eta_f)\leq
 \sqrt{(m+\eps^2/4)/2}+\sqrt{2m}\log(2m/\eps).
 \label{eq:numerical-rank-bound}
\end{equation}
Without a tail condition, the central point at $r/\eps$ is
$\eps$-accurate, with central length at most
$\norm{w}_2/\sqrt2+\sqrt r\log(r/\eps)$.
More generally, for $\eta_f\geq1$ and any $m$, with
$\tau_2^2=\sum_{i>m}w_i^2$,
\begin{equation}
 L_\CP(\eta_f)\leq\norm{w}_2/\sqrt2+
        \sqrt{m+\eta_f^2\tau_2^2/2}\log\eta_f.
 \label{eq:l2-tail-schedule}
\end{equation}
\end{corollary}
\begin{proof}
The error of the first $m$ channels is at most $m/\eta$, and that of
the tail is at most its weight sum. For $\eta\leq1$, the squared speed
is at most $\eta^2\norm{w}_2^2/2$, which gives the term
$\norm{w}_2/\sqrt2$.
For $1\leq\eta\leq\eta_f$, squared speed is at most
$m+\eta_f^2\tau_2^2/2$, proving \eqref{eq:l2-tail-schedule}.
For \eqref{eq:numerical-rank-bound}, a sharper tail estimate uses
$\min(z^2,1)\leq z$: squared speed is at most
$m+\eta\sum_{i>m}w_i\leq2m$. Also
$\norm{w}_2^2\leq m+(\sum_{i>m}w_i)^2$.
The statement without a tail condition uses $E\leq r/\eta$ and speed
at most $\sqrt r$.
\end{proof}
For $0<R<1$, by \cref{lem:chord-distance}, a central arc of length $L$
can be followed by a forward-$R$ sequence of at most
$\lceil L/\log(1+R)\rceil$ steps.
These upper bounds concern only primal movement. Consider a conic
formulation whose cone barrier restricts to the unit-scale spectral
product barrier up to an additive constant, with the same objective and
parameter $\eta$. Its projected central path is then the path studied
here, with primal length measured in the restricted Hessian metric. If
its central duality gap is $\nu/\eta$, substituting
$\eta_f=\nu/\eps$ in \eqref{eq:l2-tail-schedule} bounds the primal
length needed to reach gap $\eps$. For example, if
$\tau_2\leq\eps\sqrt m/\nu$, $w_1\leq1$, and $0<\eps\leq\min(1,\nu)$,
the primal length is at most
$\sqrt{(m+m\eps^2/\nu^2)/2}+\sqrt{3m/2}\log(\nu/\eps)$.
This bound excludes the dual movement needed to reach that gap, which
\cref{sec:primal-dual} treats.

\subsection{Explicit lower bounds and their limits}
The exact distance of \cref{thm:water-filling} dominates the
following determinant-based bound, which shows orders of growth more
directly. Let $\bar w_{g,m}=(\prod_{i\leq m}w_i)^{1/m}$.
\begin{proposition}[Determinant-based lower bound]
\label{prop:rank-certificate}
For every $0<\eps<\sum_iw_i$,
\begin{equation}
 L_\opt(\eps)\geq
 \max_{1\leq m\leq r}\sqrt m
       \left[\log\frac{m\bar w_{g,m}}{2\eps}\right]_+.
 \label{eq:rank-certificate}
\end{equation}
The bound also holds without the factor $2$, as the proof shows from
the exact distance formula. Neither form determines $L_\opt$ up to a
universal constant factor.
\end{proposition}
\begin{proof}
For any feasible allocation $x_i\in[0,1)$, put $e_i=w_i(1-x_i)$.
For a subset $I$ of size $m$, AM--GM gives
$\prod_{i\in I}(1-x_i)\leq(\eps/m)^m/\prod_{i\in I}w_i$.
Since $\rho(x)\geq\log(1/(1-x))$, Cauchy--Schwarz gives
\[
 \left(\sum_i\rho(x_i)^2\right)^{1/2}
 \geq\sqrt m\left[\log\frac{m(\prod_{i\in I}w_i)^{1/m}}{\eps}\right]_+.
\]
Minimize over allocations and choose the top $m$ weights. This proves
the stronger claim and hence \eqref{eq:rank-certificate}.
To show that the bound with the factor $2$ can vanish while
$L_\opt\to\infty$, fix
$0<c<\log(2/\sqrt e)$, $\eps=1/2$, and set
\[
 s_i=c/\sqrt i,\quad B_r=\sum_{i=1}^r i^{-1/2},\quad
 a=\eps/(cB_r),\quad w_i=a s_i e^{s_i}.
\]
The logarithmic allocation \eqref{eq:log-water-filling} has
multiplier $1/a$, optimizer $s_i$, and value
$c(\sum_i1/i)^{1/2}\to\infty$, so $L_\opt(\eps)\to\infty$ by
\eqref{eq:log-comparison}. Meanwhile
\[
 \frac{m\bar w_{g,m}}{2\eps}
 =\frac{m(m!)^{-1/(2m)}e^{(c/m)\sum_{i\leq m}i^{-1/2}}}{2B_r}
 \leq\frac{e^c\sqrt e}{2}\sqrt{m/r}<1,
\]
using $m!\geq(m/e)^m$ and $B_r\geq\sqrt r$. Thus every term of
\eqref{eq:rank-certificate} vanishes although $L_\opt\to\infty$.
For the stronger version, keep the same $c<\log(2/\sqrt e)$.
The ratio without $2$ is at most
$e^c\sqrt{er}/B_r$, uniformly in $m\leq r$.
Since $B_r\geq2(\sqrt{r+1}-1)$, this upper bound tends to
$e^c\sqrt e/2<1$. Thus even the stronger bound is identically zero for
all sufficiently large $r$.
\end{proof}

The factor $2$ comes from a determinant proof, which we give on a
matrix ball. Let the columns $u_i$ of $U$ be left singular vectors of
the objective for its $m$ largest singular values, with right singular
vectors $v_i$, and set
$\psi_U(X)=-\log\det(I-U^*XX^*U)$. Then
\begin{equation}
 \nabla^2\psi_U\preceq\nabla^2\Phi,
 \qquad |D\psi_U(X)[Z]|\leq\sqrt m\norm{Z}_{\Phi,X}.
 \label{eq:compression-certificate}
\end{equation}
Indeed, $\Phi=-\log\det\left(\begin{smallmatrix}I&X\\X^*&I\end{smallmatrix}\right)$;
$\psi_U$ is the negative log determinant of its principal compression.
Their difference is the negative log determinant of a Schur complement,
which is convex: the Schur complement is matrix concave (the quadratic
perspective $B^*A^{-1}B$ is matrix convex), and negative log determinant
is convex and order reversing. The second assertion follows from the
exact parameter $m$ of the compressed matrix-ball barrier. If
$z_i=\operatorname{Re}(u_i^*Xv_i)$, Hadamard's inequality and AM--GM
give
\[
 \det(I-U^*XX^*U)\leq\prod_{i\leq m}(1-z_i^2)
 \leq2^m\prod_{i\leq m}(1-z_i)
 \leq\frac{(2\eps/m)^m}{\prod_{i\leq m}w_i}.
\]
Integrating \eqref{eq:compression-certificate} yields
\eqref{eq:rank-certificate}. The allocation proof avoids this
compression and also applies to Jordan factors.

\subsection{Two decay laws for which the central path has optimal order}
Both examples need enough channels for the requested accuracy; for a
fixed finite rank, the asymptotics cannot hold as $\eps\downarrow0$.

For geometric weights $w_i=e^{-a(i-1)}$ with fixed $a>0$, let
$L=\log(1/\eps)$. If $L$ is sufficiently large and
$r\geq\lfloor L/a\rfloor$, then
\begin{equation}
 L_\opt(\eps)=\Theta_a(L^{3/2}),\qquad
 L_\CP(\eps)=\Theta_a(L^{3/2}).
 \label{eq:geometric-decay}
\end{equation}
To prove the lower bound, choose $m=\lfloor L/a\rfloor\geq2$ in
\eqref{eq:rank-certificate}. Its logarithm is
$L-a(m-1)/2+\log(m/2)\geq L/2$, and $m\geq L/(2a)$.
For the upper bound take
$M=\lceil(L+\log(2/(1-e^{-a})))/a\rceil$ and use
\cref{cor:numerical-rank} with $\min(M,r)$. The tail is at most
$\eps/2$ when $M<r$, and is zero otherwise; $M=O_a(L)$.
This gives length $O_a(L^{3/2})$. Thus, for these geometric weights,
the ratio $L_\CP/L_\opt$ stays bounded; geometric decay alone does not
make it grow.

For polynomial weights $w_i=i^{-b}$ with fixed $b>1$, let
$m=\lfloor(4\eps)^{-1/(b-1)}\rfloor$. If $\eps$ is sufficiently small
and $r\geq m$, then
\begin{equation}
 L_\opt(\eps)=\Theta_b(\eps^{-1/[2(b-1)]}),\qquad
 L_\CP(\eps)=\Theta_b(\eps^{-1/[2(b-1)]}).
 \label{eq:polynomial-decay}
\end{equation}
For $\eta\geq1$, splitting at $\lceil\eta^{1/b}\rceil$ and comparing
tails with integrals gives
$Q_1(\eta)\leq A_b\eta^{1/b}$ and
$Q_2(\eta)\leq B_b\eta^{1/b}$, where
$A_b=2+1/(b-1)$ and $B_b=2+1/(2b-1)$.
Consequently, the central point at $\eta_f=(A_b/\eps)^{b/(b-1)}$ is
accurate, and
\eqref{eq:Q2} bounds its length by
$\sqrt{\zeta(2b)/2}+2b\sqrt{B_b}(\eta_f^{1/(2b)}-1)$.
For the lower bound, $\bar w_{g,m}=(m!)^{-b/m}\geq m^{-b}$, so the
logarithm in \eqref{eq:rank-certificate} is at least $\log2$.
Since $m=\Theta_b(\eps^{-1/(b-1)})$, the orders agree. There is no extra
logarithmic factor because new channels activate gradually.
